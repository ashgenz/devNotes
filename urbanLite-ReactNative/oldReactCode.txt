import React, { useEffect, useRef, useState } from 'react';
  
import {
  
  View,
  
  Text,
  
  TextInput,
  
  ScrollView,
  
  Image,
  
  Dimensions,
  
  TouchableOpacity,
  
  StyleSheet,
  
} from 'react-native';
  
import { Ionicons } from '@expo/vector-icons';
  

  
const { width } = Dimensions.get('window');
  

  
const bannerImages = [
  
  require('../../assets/banner1.png'),
  
  require('../../assets/banner2.png'),
  
  require('../../assets/banner3.png'),
  
];
  

  
const HomeScreen = () => {
  
  const [location, setLocation] = useState('Pratap Nagar, Jaipur');
  
  const scrollRef = useRef<ScrollView>(null);
  
  const [currentIndex, setCurrentIndex] = useState(0);
  

  
  // Auto-slide banner logic
  
  useEffect(() => {
  
    const interval = setInterval(() => {
  
      const nextIndex = (currentIndex + 1) % bannerImages.length;
  
      scrollRef.current?.scrollTo({ x: nextIndex * width, animated: true });
  
      setCurrentIndex(nextIndex);
  
    }, 3000); // 3 seconds
  

  
    return () => clearInterval(interval);
  
  }, [currentIndex]);
  

  
  return (
  
    <View style={styles.container}>
  
      {/* Top Bar */}
  
      <View style={styles.topBar}>
  
        <Text style={styles.logoText}>UrbanLite</Text>
  
        <View style={styles.topIcons}>
  
          <Ionicons name="walk" size={24} color="white" style={{ marginRight: 12 }} />
  
          <View style={styles.profileIcon} />
  
        </View>
  
      </View>
  

  
      <View style={styles.innerContainer}>
  
        {/* Search Fields */}
  
        <View style={styles.searchContainer}>
  
          <TextInput style={styles.input} placeholder="Maid" />
  
          <TextInput
  
            style={styles.input}
  
            value={location}
  
            onChangeText={setLocation}
  
            placeholder="Enter location"
  
          />
  
        </View>
  

  
        {/* Auto-sliding Banner */}
  
        <ScrollView
  
          ref={scrollRef}
  
          horizontal
  
          pagingEnabled
  
          showsHorizontalScrollIndicator={false}
  
          style={styles.banner}
  
        >
  
          {bannerImages.map((img, index) => (
  
            <Image key={index} source={img} style={styles.bannerImage} />
  
          ))}
  
        </ScrollView>
  

  
        {/* Service Selection */}
  
        <Text style={styles.sectionTitle}>Select a service</Text>
  
        <View style={styles.boxRow}>
  
          <Box label="Cook" />
  
          <Box label="Cleaner" />
  
          <Box label="Driver" />
  
        </View>
  

  
        {/* Plan Selection */}
  
        <Text style={styles.sectionTitle}>Select a plan</Text>
  
        <View style={styles.boxRow}>
  
          <Box label="Standard" />
  
          <Box label="Premium" />
  
          <Box label="Monthly" />
  
        </View>
  

  
        {/* Schedule */}
  
        <Text style={styles.sectionTitle}>Schedule</Text>
  
        <TextInput style={styles.input} placeholder="Choose time" />
  

  
        {/* Proceed Button */}
  
        <TouchableOpacity style={styles.proceedButton}>
  
          <Text style={styles.proceedText}>Proceed</Text>
  
        </TouchableOpacity>
  
      </View>
  
    </View>
  
  );
  
};
  

  
const Box = ({ label }: { label: string }) => (

    <Text>{label}</Text>
  
  </View>
  
);
